        // 渲染測驗

        function renderQuiz() {

            const container = document.getElementById('quiz-container');

            container.innerHTML = '';


            quizData.forEach((q, qIndex) => {

                let optionsHTML = '';

                q.options.forEach((opt, oIndex) => {

                    optionsHTML += `

                        <div class="card p-3 mb-2 quiz-option" id="opt-${qIndex}-${oIndex}" onclick="checkAnswer(${qIndex}, ${oIndex})">

                            ${opt}

                        </div>

                    `;

                });


                container.innerHTML += `

                    <div class="card p-4 mb-4 shadow-sm">

                        <h5 class="fw-bold mb-3">${q.question}</h5>

                        <div>${optionsHTML}</div>

                        <div class="alert alert-info mt-3 d-none" id="explain-${qIndex}">

                            ${q.explain}

                        </div>

                    </div>

                `;

            });

        }


        // 驗證答案

        function checkAnswer(qIndex, selectedIndex) {

            const q = quizData[qIndex];

            const explainDiv = document.getElementById(`explain-${qIndex}`);

            

            // 防止重複點選

            const options = document.querySelectorAll(`[id^="opt-${qIndex}-"]`);

            let isAnswered = false;

            options.forEach(opt => {

                if(opt.classList.contains('correct-bg') || opt.classList.contains('incorrect-bg')) {

                    isAnswered = true;

                }

            });

            if(isAnswered) return;


            // 標示正確與錯誤選項

            options.forEach((opt, oIndex) => {

                if(oIndex === q.answer) {

                    opt.classList.add('correct-bg');

                }

                if(oIndex === selectedIndex && selectedIndex !== q.answer) {

                    opt.classList.add('incorrect-bg');

                }

            });


            // 計算分數

            if(selectedIndex === q.answer) {

                currentScore += 20;

            }

            answeredCount++;

            

            // 顯示得分與解析

            document.getElementById('score-display').innerText = `得分：${currentScore} / ${quizData.length * 20}`;

            explainDiv.classList.remove('d-none');

        }

        document.addEventListener('DOMContentLoaded', renderQuiz);
